#!/usr/bin/env python3
"""Nightly build for the online Daybook (tablet copy).
Usage: python3 -I build.py WORKDIR EXISTING_THUMBS_TXT
WORKDIR holds daybook_fetch.py, channels.json, topics.json, cache.json.
Writes WORKDIR/out/: feed.js, cache.json, new_thumbs.txt, stale_thumbs.txt."""
import json, os, subprocess, sys, shutil
w = os.path.abspath(sys.argv[1]); existing_file = sys.argv[2] if len(sys.argv) > 2 else None
os.makedirs(os.path.join(w, "icons"), exist_ok=True)
os.makedirs(os.path.join(w, "thumbs"), exist_ok=True)
# icons already live in the online page: placeholders stop the fetcher re-fetching them
for c in json.load(open(os.path.join(w, "channels.json"), encoding="utf-8")):
    if c.get("id") and c.get("icon"):
        p = os.path.join(w, "icons", c["id"] + ".jpg")
        if not os.path.exists(p):
            open(p, "wb").close()
r = subprocess.run([sys.executable, "-I", os.path.join(w, "daybook_fetch.py")], cwd=w, capture_output=True, text=True, timeout=1500)
print(r.stdout[-3000:]); print(r.stderr[-2000:])
if r.returncode != 0:
    sys.exit("fetch failed")
out = os.path.join(w, "out"); os.makedirs(out, exist_ok=True)
shutil.copy(os.path.join(w, "feed.js"), out); shutil.copy(os.path.join(w, "cache.json"), out)
have = set()
if existing_file and os.path.exists(existing_file):
    have = {l.strip() for l in open(existing_file) if l.strip()}
now = {"thumbs/" + f for f in os.listdir(os.path.join(w, "thumbs")) if f.endswith(".jpg")}
open(os.path.join(out, "new_thumbs.txt"), "w").write("\n".join(sorted(now - have)))
open(os.path.join(out, "stale_thumbs.txt"), "w").write("\n".join(sorted(have - now)))
print("new thumbs:", len(now - have), "stale:", len(have - now))
